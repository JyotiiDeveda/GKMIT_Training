-- Query
-- 1. Get all the users that are associated to only one city.
	select
		users.id,
		users.name
	from
		users 
	join
		user_cities on users.id=user_cities.user_id
	group by
		users.id
	having
		count(users.id) = 1;


	-- using subquery
	select 
	    users.name
	from 
	    users 
	join (
	    select 
	       user_id, 
	       count(user_id) as user_cnt 
	    from 
	       user_cities 
	    group by user_id 
	    having 
	       count(user_id) = 1) 
	    as user_city on users.id=user_city.user_id;



-- 2. Get all the users that are associated to only one city and that is 'Udaipur'.

	-- using aggregation
	select 
		* 
	from(
		select
			users.id,
			users.name,
			string_agg(
				user_cities.city,
				','
			) cityname
		from
			users 
		join
			user_cities on users.id=user_cities.user_id
		group by
			users.id
	)
	where 
		cityname='Udaipur';


	-- using subquery
	select
		users.name, 
	  	user_cities.city
	from 
		users
	join
		user_cities
	on
		users.id=user_cities.user_id
	where 
		users.id in
	  (
	    select 
	      uc.user_id
	    from 
	      user_cities uc
	    group by
	      uc.user_id
	    having 
	      count(uc.user_id)=1
	  ) and user_cities.city = 'Udaipur';
